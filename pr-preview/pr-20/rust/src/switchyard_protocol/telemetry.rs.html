4//! Provider-neutral telemetry vocabulary shared across Switchyard crates.
5//!
6//! This module defines the stable string contract for route-selection and
7//! Langfuse observations: generic OpenTelemetry attribute keys, Langfuse-native
8//! metadata keys, and the allowed values for response origin and call role. It
9//! carries only names and values: no instrumentation, exporter, or metric
10//! behavior, so any crate can emit or read the same vocabulary without
11//! depending on a telemetry backend.
12
13use std::fmt;
14
15/// Generic OpenTelemetry attribute keys for route-selection telemetry.
16///
17/// Backend-neutral keys an exporter maps onto its own attribute namespace.
18/// They are associated `const`s so callers may use them in attribute tables
19/// and span metadata without allocation.
20pub struct GenericKeys;
21
22impl GenericKeys {
23    /// Route the request was sent for.
24    pub const ROUTE_ID: &str = "switchyard.route.id";
25    /// Routing algorithm that produced the selection.
26    pub const ALGORITHM: &str = "switchyard.routing.algorithm";
27    /// Target the routing algorithm selected.
28    pub const SELECTED_TARGET: &str = "switchyard.routing.selected_target";
29    /// Target that actually served the response.
30    pub const SERVED_TARGET: &str = "switchyard.response.served_target";
31    /// Origin of the served response, one of [`ResponseOrigin`].
32    pub const RESPONSE_ORIGIN: &str = "switchyard.response.origin";
33    /// Role of the call within the route, one of [`CallRole`].
34    pub const CALL_ROLE: &str = "switchyard.call.role";
35    /// Target the call was directed at.
36    pub const CALL_TARGET: &str = "switchyard.call.target";
37}
38
39/// Langfuse-native metadata keys for route-selection observations.
40///
41/// The literal key strings Langfuse expects on an observation's `metadata`
42/// object. `const`s so callers can build observation metadata without
43/// allocation.
44pub struct LangfuseKeys;
45
46impl LangfuseKeys {
47    /// Session identifier that groups an observation.
48    pub const SESSION_ID: &str = "langfuse.session.id";
49    /// User identifier attributed to an observation.
50    pub const USER_ID: &str = "langfuse.user.id";
51    /// Observation metadata key for the route id.
52    pub const ROUTE_ID: &str = "langfuse.observation.metadata.switchyard.route_id";
53    /// Observation metadata key for the routing algorithm.
54    pub const ALGORITHM: &str = "langfuse.observation.metadata.switchyard.algorithm";
55    /// Observation metadata key for the selected target.
56    pub const SELECTED_TARGET: &str = "langfuse.observation.metadata.switchyard.selected_target";
57    /// Observation metadata key for the served target.
58    pub const SERVED_TARGET: &str = "langfuse.observation.metadata.switchyard.served_target";
59    /// Observation metadata key for the response origin.
60    pub const RESPONSE_ORIGIN: &str = "langfuse.observation.metadata.switchyard.response_origin";
61    /// Observation metadata key for the call role.
62    pub const CALL_ROLE: &str = "langfuse.observation.metadata.switchyard.call_role";
63    /// Observation metadata key for the call target.
64    pub const CALL_TARGET: &str = "langfuse.observation.metadata.switchyard.call_target";
65}
66
67/// Origin of a served response.
68#[derive(Clone, Copy, Debug, Eq, PartialEq)]
69pub enum ResponseOrigin {
70    /// Served by a backing target model.
71    Target,
72    /// Generated by Switchyard's routing layer rather than a target model.
73    RoutingGenerated,
74}
75
76impl ResponseOrigin {
77    /// Returns the stable contract value for the origin.
78    pub const fn as_str(self) -> &'static str {
79        match self {
80            Self::Target => "target",
81            Self::RoutingGenerated => "routing_generated",
82        }
83    }
84}
85
86impl fmt::Display for ResponseOrigin {
87    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
88        formatter.write_str(self.as_str())
89    }
90}
91
92/// Role a call plays within a routed request.
93#[derive(Clone, Copy, Debug, Eq, PartialEq)]
94pub enum CallRole {
95    /// A call Switchyard makes while selecting a route, not for the answer.
96    RoutingDependency,
97    /// The primary model serving the answer.
98    AnswerPrimary,
99    /// A fallback model serving the answer.
100    AnswerFallback,
101}
102
103impl CallRole {
104    /// Returns the stable contract value for the role.
105    pub const fn as_str(self) -> &'static str {
106        match self {
107            Self::RoutingDependency => "routing_dependency",
108            Self::AnswerPrimary => "answer_primary",
109            Self::AnswerFallback => "answer_fallback",
110        }
111    }
112}
113
114impl fmt::Display for CallRole {
115    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
116        formatter.write_str(self.as_str())
117    }
118}
